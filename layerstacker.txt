'use client';

import { useRef, useState } from 'react';
import registry from './registry.json';

const LAYER_ORDER = [
  'Background', 'Base Body', 'Face', 'Eye', 
  'Outfits', 'Jeweleries', 'Eyewear', 'Headwear'
];

export function LayerStacker() {
  const [currentIndex, setCurrentIndex] = useState(0);
  const [isExporting, setIsExporting] = useState(false);
  const canvasRef = useRef<HTMLCanvasElement>(null);

  const operative = registry[currentIndex];

  if (!operative) return <div style={{ color: '#A3A3A3' }}>Loading Operative Data...</div>;

  // Helper to safely build URLs
  const getSafeUrl = (layerName: string) => {
    // For local Vite testing, we pull the clean name (e.g., "Bright Silver") 
    // from display_traits to completely bypass the '#' bug.
    const cleanTraitName = (operative.display_traits as Record<string, string>)[layerName];
    
    if (!cleanTraitName || cleanTraitName === 'None') return null;

    const safeLayer = encodeURIComponent(layerName);
    const safeFile = encodeURIComponent(`${cleanTraitName}.png`);

    // --- LOCAL DEV MODE ---
    return `/layers/${safeLayer}/${safeFile}`;

    // --- MAINNET MODE (Swap to this when you launch!) ---
    // const SITE_ID = 'YOUR_MAINNET_SITE_ID';
    // const walrusFilename = (operative.walrus_files as Record<string, string>)[layerName === 'Jewelry' ? 'Jeweleries' : layerName];
    // return `https://${SITE_ID}.wal.app/${safeLayer}/${encodeURIComponent(walrusFilename)}`;
  };

  const exportCompositeImage = async () => {
    setIsExporting(true);
    const canvas = canvasRef.current || document.createElement('canvas');
    canvas.width = 2000;
    canvas.height = 2000;
    const ctx = canvas.getContext('2d');

    if (!ctx) {
      setIsExporting(false);
      return;
    }
    ctx.clearRect(0, 0, 2000, 2000);

    const loadImage = (src: string): Promise<HTMLImageElement> => {
      return new Promise((resolve, reject) => {
        const img = new Image();
        img.crossOrigin = 'anonymous'; 
        img.onload = () => resolve(img);
        img.onerror = () => reject(new Error(`Failed to load: ${src}`));
        img.src = src;
      });
    };

    try {
      for (const layerName of LAYER_ORDER) {
        const url = getSafeUrl(layerName);
        if (!url) continue;
        const img = await loadImage(url);
        ctx.drawImage(img, 0, 0, 2000, 2000);
      }

      const dataUrl = canvas.toDataURL('image/png');
      const downloadLink = document.createElement('a');
      downloadLink.href = dataUrl;
      downloadLink.download = `numb_operative_${operative.id}.png`;
      downloadLink.click();
    } catch (err) {
      console.error('Compositing failed:', err);
      alert('Could not export composite. Ensure files exist locally.');
    } finally {
      setIsExporting(false);
    }
  };

  return (
    <div style={{ display: 'flex', flexDirection: 'column', alignItems: 'center', width: '100%' }}>
      
      {/* LORE IDENTIFIER */}
      <div style={{ textAlign: 'center', marginBottom: '16px', width: '100%' }}>
        <h2 style={{ fontSize: '14px', fontWeight: 'bold', color: '#06B6D4', textTransform: 'uppercase', margin: '0 0 4px 0', letterSpacing: '1px' }}>
          {operative.lore_name}
        </h2>
        <p style={{ color: '#888', fontSize: '11px', textTransform: 'uppercase', margin: 0, letterSpacing: '0.5px' }}>
          TIER: {operative.display_traits['Rarity Tier']} | ID: #{operative.id}
        </p>
      </div>

      {/* PREVIEW CONTAINER */}
      <div
        style={{
          position: 'relative',
          width: '380px',
          height: '380px',
          background: 'rgba(13, 13, 17, 0.9)',
          border: '1px solid rgba(255, 255, 255, 0.15)',
          backdropFilter: 'blur(12px)',
          overflow: 'hidden',
          marginBottom: '1rem',
          borderRadius: '2px',
          boxShadow: '0 0 30px rgba(6, 182, 212, 0.08)',
        }}
      >
        {LAYER_ORDER.map((layerName, index) => {
          const url = getSafeUrl(layerName);
          if (!url) return null;
          return (
            <img
              key={layerName}
              src={url}
              alt={layerName}
              crossOrigin="anonymous"
              style={{
                position: 'absolute',
                top: 0,
                left: 0,
                width: '100%',
                height: '100%',
                objectFit: 'contain',
                zIndex: index,
              }}
            />
          );
        })}
      </div>

      {/* CONTROLS */}
      <div style={{ display: 'flex', gap: '8px', width: '380px', marginBottom: '8px' }}>
        <button 
          onClick={() => setCurrentIndex(prev => Math.max(0, prev - 1))}
          style={{ flex: 1, padding: '10px', background: 'rgba(255,255,255,0.03)', color: '#E5E5E5', border: '1px solid rgba(255, 255, 255, 0.15)', cursor: 'pointer', fontFamily: 'inherit', fontSize: '11px', textTransform: 'uppercase', letterSpacing: '0.1em' }}
        >
          ← PREV
        </button>
        <button 
          onClick={() => setCurrentIndex(prev => Math.min(registry.length - 1, prev + 1))}
          style={{ flex: 1, padding: '10px', background: 'rgba(255,255,255,0.03)', color: '#E5E5E5', border: '1px solid rgba(255, 255, 255, 0.15)', cursor: 'pointer', fontFamily: 'inherit', fontSize: '11px', textTransform: 'uppercase', letterSpacing: '0.1em' }}
        >
          NEXT →
        </button>
      </div>

      {/* EXPORT BUTTON */}
      <button
        onClick={exportCompositeImage}
        disabled={isExporting}
        style={{
          display: 'block',
          width: '380px',
          padding: '12px',
          background: isExporting ? 'rgba(255,255,255,0.05)' : 'rgba(255, 255, 255, 0.03)',
          color: isExporting ? '#777' : '#E5E5E5',
          border: '1px solid rgba(255, 255, 255, 0.15)',
          cursor: isExporting ? 'not-allowed' : 'pointer',
          fontFamily: 'inherit',
          fontWeight: '500',
          fontSize: '11px',
          textTransform: 'uppercase',
          letterSpacing: '0.1em',
          borderRadius: '2px',
          transition: 'all 0.3s ease',
        }}
      >
        {isExporting ? 'COMPOSITING 2000x2000...' : '⬇ EXPORT COMPOSITE PNG'}
      </button>
    </div>
  );
}
